\section{Starting point}
% ===========================================================================

\begin{frame}{The problem}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item Skipper-CCD read out with an \textbf{LTA} controller
        \item Find the operating parameters that give the best images
        \item Each evaluation is a \textbf{real measurement}:
              expose (LED) $\rightarrow$ set parameters $\rightarrow$ read out
              $\rightarrow$ score
        \item Expensive, noisy, black-box $\Rightarrow$ Bayesian optimization
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering\small
      \begin{tabular}{lrl}
        \toprule
        Parameter & Range & \\
        \midrule
        $V_{dd}$         & $[-23,-10]$ & V \\
        $V_{drain}$      & $[-23,-10]$ & V \\
        $V_{r}$          & $[-8,-5]$   & V \\
        \texttt{delay\_H\_overlap} & $[10,30]$ & clk \\
        \bottomrule
      \end{tabular}

      \vspace{2mm}
      {\scriptsize Optimizer works in $[-1,1]^4$; values mapped to physical units.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The objective function}
  From each image: active region and overscan region
  \[
    \text{gain} = \operatorname{median}(\text{active}) - \operatorname{median}(\text{overscan}),
    \qquad
    \sigma_{os} = \operatorname{std}(\text{overscan})
  \]
  \[
    F = \min\!\left(\left|\frac{\sigma_{os}}{\text{gain}}\right|,\ 1000\right)
        \;+\; \sum_i c_i\,\text{penalty}_i^{\,p_i}
    \qquad\text{(lower is better)}
  \]
  \begin{itemize}
    \item With a fixed LED exposure, gain $\propto$ ADU/e$^-$, so $F$ tracks the
          read noise in electrons
    \item Optional penalties (negative gain, saturation, \dots) are configurable;
          all off in the current campaign
  \end{itemize}
\end{frame}

\section{Review}
% ===========================================================================

\begin{frame}{Code review: things to watch}
  \begin{itemize}
    \item \textbf{Stale images}: shell return codes ignored and the newest
          \texttt{.fz} is scored, so a failed readout would be scored using
          the previous image
    \item \textbf{Bad amplifier states can score well}: \texttt{abs()} hides
          an inverted gain; a clipped output has $\sigma_{os}\to 0$
    \item \textbf{Real example} from a lab run:
          \begin{center}\small
          \begin{tabular}{cccc|ccc}
            $V_{dd}$ & $V_{drain}$ & $V_r$ & delay & $\sigma_{os}$ & gain & $F$\\\midrule
            $-10.3$ & $-12.5$ & $-5.2$ & 15 & 31924 & $-3544$ & 9.0
          \end{tabular}\end{center}
          a broken output still gets a finite, ``reasonable'' score
    \item \textbf{No settling time} after changing bias voltages
    \item \textbf{Noisy $F$}: the best \emph{measured} point may be a lucky draw
    \item \textbf{Resume file} only written at the end of a run
  \end{itemize}
  \vspace{1mm}
  {\scriptsize Kept as-is on purpose (no functional changes); the resume issue was fixed later.}
\end{frame}

\begin{frame}{First lab deployment: one surprise}
  \begin{itemize}
    \item Error: \emph{``No .fz image found''}, but the LTA said \emph{``Read done''}
    \item Cause: relative output path \texttt{images/...}
      \begin{itemize}\small
        \item Python resolved it from \emph{its} working directory
        \item the LTA daemon resolved it from \emph{its own} directory
        \item $\Rightarrow$ images written to one folder, searched for in another
      \end{itemize}
    \item Fix: absolute \texttt{output\_base} in the config (no code change)
    \item Not specific to agents: the original script had the same latent issue
  \end{itemize}
\end{frame}

\section{Claude as optimizer}
% ===========================================================================

\begin{frame}[fragile]{Step 2: Claude replaces the Gaussian process}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{enumerate}\small
        \item \textbf{Sobol start}: 8 space-filling points (as for the GP)
        \item \textbf{Each further point}: Claude gets
          \begin{itemize}\scriptsize
            \item parameter descriptions and bounds, objective definition
            \item \emph{all} measurements so far, in physical units
            \item image statistics ($\sigma_{os}$, $\sigma_{act}$, charges, gain),
                  which the GP never sees
            \item operator notes and the lab notebook
          \end{itemize}
        \item Claude returns the next point as \textbf{structured JSON};
              values clipped to bounds
      \end{enumerate}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Drop-in replacement}
        \scriptsize
        \texttt{claude\_minimize()} has the same interface and returns the same
        result object as \texttt{gp\_minimize()}: progress CSV, resume, pickle
        and convergence plot unchanged.
      \end{block}
      \vspace{2mm}

  "effort": "high",

\end{lstlisting}

\end{frame}

\begin{frame}{Three campaigns, three amplifiers (Oct 1, 2026)}
  \small
  Same config: 30 images each, the same 8 Sobol starting points, then 22 chosen by Claude
  (Opus 5.5); the lab notebook carried over from one campaign to the next.

  \vspace{2mm}
  \begin{tabular}{llllll}
    \toprule
    & Amplifier & Best $F$ & Best point ($V_{dd}, V_{drain}, V_r$, delay) & $\sigma_{os}$ (ADU) & Duration \\
    \midrule
    Campaign 1 & \ampA & 0.0189 & $(-14.5,\,-18.8,\,-6.3,\,21)$ & $\approx 700$ & 12 min \\
    Campaign 2 & \ampB & 0.0242 & $(-17.5,\,-20.2,\,-5.0,\,30)$ & $\approx 700$ & 12 min \\
    Campaign 3 & \ampC & 1.13   & $(-16.9,\,-19.6,\,-7.0,\,22)$ & $3$--$4.5\times10^{4}$ & 11 min \\
    \bottomrule
  \end{tabular}

  \vspace{3mm}
  \begin{itemize}
    \item Campaigns 1 and 2: working amplifiers, $F \approx 0.02$--$0.03$ in the good region;
          repeats at a fixed setting scatter by about $\pm 0.005$
    \item \textbf{Campaign 3: amplifier 0 not working.} Overscan noise $\sim$50$\times$ higher,
          overscan level $\approx -4.8\times10^{5}$ ADU; $F$ never below 1.1. Amplifier 0 shows
          $\sigma_{os} \approx 3\times10^{4}$ ADU in every image, also in the other campaigns' images
    \item Repeatable failures: the shallow corner $(-10.3, -12.5, -5.2)$ in all three
          campaigns; $V_{drain}$ less negative than $V_{dd}$ in campaigns 1 and 2
  \end{itemize}

  \vspace{1mm}
  {\scriptsize Amplifiers numbered 0--3; the config's \texttt{amplifier} value is the FITS HDU
   index, one higher (HDU 0 is the empty primary). Identified by recomputing $F$ for every
   extension and matching the logged value.}

\section{Lab notebook}
% ===========================================================================

\begin{frame}{Lab notebook: memory across campaigns}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \small
      \begin{itemize}
        \item Each API call starts from scratch: no learning between runs
        \item \textbf{End of campaign}: Claude writes a short entry
              (best region, broken settings, sensitivities, suggestions)
        \item \textbf{Start of next campaign}: notebook loaded into Claude's
              instructions as \emph{prior knowledge}
        \item Current measurements take priority over old entries
        \item Plain Markdown file: operators read, edit, add their own notes
        \item Header line written by the code (date, module, best point);
              text by Claude
      \end{itemize}
    \end{column}
    \begin{column}{0.52\textwidth}
      \scriptsize
      \begin{block}{Entry 1 (campaign 1), excerpt}
        \emph{Measurements \#11--18 all had near-zero or negative gain. This suggests an
        episode where the LED or signal was lost, rather than a parameter effect.}\\[1mm]
        \emph{Suggestions: start at (-14.5, -18.8, -6.3, 21) [\dots] interleave periodic
        re-measurements of a reference point to detect LED or signal dropouts.}
      \end{block}
      \vspace{1mm}
      \begin{block}{Entry 2 (campaign 2), excerpt}
        \emph{Last campaign's best point (-14.5, -18.8, -6.3, 21) gave only 0.047 this
        time. Its earlier 0.019 was probably a lucky draw or reflects changed conditions.}\\[1mm]
        \emph{Use at least 3 repeats per candidate. Single-shot differences below about
        0.01 are not meaningful.}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Lab notebook across amplifiers: a lesson}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \scriptsize
      \begin{block}{Entry 3 (campaign 3), excerpt}
        \emph{Setup state changed sharply this campaign. noise\_overscan was 3.1--4.5e4 ADU,
        about 50$\times$ the $\sim$700 ADU of earlier campaigns. [\dots] F values from this
        campaign are not comparable with earlier entries. The likely cause is a hardware,
        offset or cabling change rather than a parameter effect, but this is unconfirmed.}\\[1mm]
        \emph{Before optimizing, measure a reference point and check noise\_overscan and
        charge\_overscan. [\dots] investigate the readout chain or offset first.}
      \end{block}
    \end{column}
    \begin{column}{0.47\textwidth}
      \small
      \begin{itemize}
        \item Each campaign used a \textbf{different amplifier}, but the notebook was
              shared and the amplifier was not in it
        \item Claude correctly detected that conditions had changed, and that the
              old best points no longer held
        \item It could only guess the cause (``hardware, offset or cabling'')
        \item The third amplifier was in fact not working: the right call was to stop
              optimizing and check the readout chain, as the entry says
      \end{itemize}
      \vspace{1mm}
      \begin{block}{Improvement}
        \scriptsize Record the amplifier in each entry, or keep one notebook per
        amplifier (\texttt{"notebook"} path in the config).
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
